import struct

def encode_varint(value: int) -> bytes:
    if not isinstance(value, int) or isinstance(value, bool):
        raise ValueError("value must be an integer")
    if value < 0 or value > 2**64 - 1:
        raise ValueError("value out of range for unsigned 64-bit varint")
    
    result = bytearray()
    while True:
        byte = value & 0x7F
        value >>= 7
        if value:
            byte |= 0x80
        result.append(byte)
        if not value:
            break
    return bytes(result)


def decode_varint(data: bytes, offset: int = 0) -> tuple:
    if not isinstance(data, (bytes, bytearray)):
        raise ValueError("data must be bytes")
    if not isinstance(offset, int) or isinstance(offset, bool):
        raise ValueError("offset must be an integer")
    if offset < 0:
        raise ValueError("offset must be nonnegative")
    if offset >= len(data):
        raise ValueError("offset out of range")
    
    value = 0
    shift = 0
    num_bytes = 0
    pos = offset
    
    while True:
        if pos >= len(data):
            raise ValueError("truncated varint")
        byte = data[pos]
        pos += 1
        num_bytes += 1
        
        if num_bytes > 10:
            raise ValueError("varint exceeds 10 bytes")
        
        value |= (byte & 0x7F) << shift
        shift += 7
        
        if not (byte & 0x80):
            break
    
    # Check for overflow: value must fit in 64 bits
    if value > 2**64 - 1:
        raise ValueError("varint overflow")
    
    # Check for noncanonical encoding: the last byte should not be 0x00
    # unless the varint is exactly one byte (i.e., value is 0)
    # A noncanonical encoding has a trailing 0x00 byte when it's not the only byte
    # More precisely: if num_bytes > 1 and the last byte is 0x00, it's noncanonical
    # because the value could be represented with fewer bytes.
    last_byte = data[pos - 1]
    if num_bytes > 1 and last_byte == 0x00:
        raise ValueError("noncanonical varint encoding")
    
    return (value, pos)
